Load data:

library(dplyr)
# Load data about visits to national parks in 2020
np_visits_2020_df <- read.csv("https://raw.githubusercontent.com/melaniewalsh/Neat-Datasets/refs/heads/main/2020-National-Park-Visits-By-State.csv")
# Load data about state populations in 2020 (based on https://www.census.gov/data/tables/time-series/demo/popest/2020s-state-total.html)
state_pop_2020_df <- read.csv("https://raw.githubusercontent.com/kylethayer/Info-201-au-24-datasets/refs/heads/main/NST-EST2020-POP.csv")

# Part 1: Get visit information by state

Visits per state:

# use group_by and summarize to make a new dataframe called np_state_summary_df with:
# - State
# - total_parks (total number of state parks in the state)
# - total_visits (total number of RecreationVisits to parks in the state)
np_state_summary_df <-
  np_visits_2020_df |>
  group_by(State) |>
  summarize(
    total_parks = n(),
    total_visits = sum(RecreationVisits)
  )
np_state_summary_df

States with the most visits:

# Find the row(s) for the state(s) with the most visits
# then use pull to get the State(s) names out
# save as max_np_visit_state
max_np_visit_state <-
  np_state_summary_df |>
  filter(total_visits == max(total_visits)) |>
  pull(State)

Making 2 new rows:

# Use mutate to make two new rows: 
# - np_per_capita (the total_parks divided by pop_2020)
# - np_visits_per_capita (the total_visits divided by pop_2020)
# Save back into the np_state_pop_df variable
np_state_pop_df <- 
  np_state_pop_df |>
  mutate(
    np_per_capita = total_parks / pop_2020,
    np_visits_per_capita = total_visits / pop_2020
